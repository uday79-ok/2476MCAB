Views canteen/views.py python Copy Edit from django.shortcuts import render, redirect, get_object_or_404 from django.contrib.auth.decorators import login_required from django.contrib.admin.views.decorators import staff_member_required from django.contrib import messages from .models import FoodItem, Order def menu(request): items = FoodItem.objects.all() return render(request, 'canteen/menu.html', {'items': items}) @login_required def place_order(request, item_id): item = get_object_or_404(FoodItem, id=item_id) if request.method == 'POST': quantity = int(request.POST.get('quantity', 1)) Order.objects.create(user=request.user, food_item=item, quantity=quantity) return redirect('view_orders') return render(request, 'canteen/place_order.html', {'item': item}) @login_required def view_orders(request): orders = Order.objects.filter(user=request.user) return render(request, 'canteen/view_orders.html', {'orders': orders}) @login_required def make_payment(request, order_id): order = get_object_or_404(Order, id=order_id, user=request.user) if not order.is_paid: order.is_paid = True order.status = 'PREPARING' order.save() messages.success(request, "Payment successful!") else: messages.info(request, "Order already paid.") return redirect('view_orders') @staff_member_required def all_orders(request): orders =
